# Font Size Optimization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Reduce Android .so binary from 24MB to ~9MB through font subsetting and platform-specific asset loading

**Architecture:** Hybrid platform optimization - Android loads fonts from APK assets at runtime, desktop embeds subsetted icon font via include_bytes!, build.rs subsets Material Symbols font using pyftsubset

**Tech Stack:** Rust, pyftsubset (fonttools), egui, ndk::asset::AssetManager, cargo-ndk

**Spec:** `docs/superpowers/specs/2026-08-28-font-optimization-design.md`

## Global Constraints

- Python 3.x + fonttools (`pip install fonttools`) required for build
- Material Symbols subset must contain exactly 6 icons: SETTINGS, MENU, INFO, STAR, STAR_OUTLINE, BLOCK
- Android .so target size: ~9MB (down from 24MB)
- Desktop binary target size: ~15MB (down from 24MB)
- All existing UI functionality must remain unchanged
- Build must fail-fast with clear error if pyftsubset unavailable

// mobile/build.rs
use std::process::Command;
use std::path::Path;
use anyhow::{Context, Result};

fn main() -> Result<()> {
    println!("cargo:rerun-if-changed=resources/MaterialSymbolsOutlined[FILL,GRAD,opsz,wght].ttf");
    
    // 1. Subset Material Symbols font
    subset_material_symbols()?;
    
    // 2. Copy fonts to Android assets (only when building for Android)
    #[cfg(target_os = "android")]
    {
        copy_to_android_assets("MaterialSymbolsOutlined-subset.ttf")?;
        copy_to_android_assets("noto-sans-kr.ttf")?;
    }
    
    // 3. Windows icon (existing functionality)
    #[cfg(target_os = "windows")]
    {
        let mut res = winres::WindowsResource::new();
        res.set_icon("resources/logo.ico");
        res.compile()?;
    }
    
    Ok(())
}

fn subset_material_symbols() -> Result<()> {
    let input = "resources/MaterialSymbolsOutlined[FILL,GRAD,opsz,wght].ttf";
    let output = "resources/MaterialSymbolsOutlined-subset.ttf";
    
    // Icon Unicode codepoints (from Task 1 - REPLACE WITH ACTUAL VALUES)
    // TODO: Replace these placeholder values with actual values from Task 1
    let unicodes = "U+e5d2,U+e8b8,U+e88e,U+e838,U+e83a,U+e14b";
    
    println!("Subsetting Material Symbols font: {} icons", 6);
    
    let status = Command::new("pyftsubset")
        .arg(input)
        .arg(format!("--unicodes={}", unicodes))
        .arg(format!("--output-file={}", output))
        .status()
        .context("Failed to run pyftsubset - is it installed? Run: pip install fonttools")?;
    
    if !status.success() {
        anyhow::bail!("Font subsetting failed");
    }
    
    // Verify output file exists and is not empty
    let subset_path = Path::new(output);
    if !subset_path.exists() {
        anyhow::bail!("Subset font was not created at {}", output);
    }
    
    let metadata = std::fs::metadata(subset_path)
        .context("Failed to read subset font metadata")?;
        
    if metadata.len() < 1000 {
        anyhow::bail!("Subset font suspiciously small ({}bytes) - subsetting may have failed", metadata.len());
    }
    
    if metadata.len() > 200_000 {
        anyhow::bail!("Subset font too large ({}bytes) - should be ~50KB", metadata.len());
    }
    
    println!("Font subsetting successful: {} bytes", metadata.len());
    Ok(())
}

#[cfg(target_os = "android")]
fn copy_to_android_assets(filename: &str) -> Result<()> {
    let src = Path::new("resources").join(filename);
    let dest = Path::new("app/src/main/assets/fonts").join(filename);
    
    // Create assets directory if it doesn't exist
    std::fs::create_dir_all(dest.parent().unwrap())
        .context("Failed to create assets/fonts directory")?;
    
    std::fs::copy(&src, &dest)
        .context(format!("Failed to copy {} to Android assets", filename))?;
    
    println!("Copied {} to Android assets", filename);
    Ok(())
}

#[cfg(not(target_os = "android"))]
fn copy_to_android_assets(_filename: &str) -> Result<()> {
    // No-op on non-Android platforms
    Ok(())
}

use std::io::Read;

fn load_font_from_assets(asset_manager: &ndk::asset::AssetManager, path: &str) -> anyhow::Result<Vec<u8>> {
    let mut asset = asset_manager.open(path)
        .context(format!("Failed to open asset: {}", path))?;
    
    let mut bytes = Vec::new();
    asset.read_to_end(&mut bytes)
        .context("Failed to read font asset")?;
    
    Ok(bytes)
}
